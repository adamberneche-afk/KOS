function openBragBoard(autoPrompt) {
  // Set week-start to most recent Monday
  const d = new Date();
  const day = d.getDay();
  const diff = (day === 0) ? -6 : 1 - day;
  d.setDate(d.getDate() + diff);
  const iso = d.toISOString().slice(0, 10);
  document.getElementById('brag-week-start').value = iso;

  setBragTab(_bragTab, document.getElementById('brag-tab-' + _bragTab));
  gatherBragData();
  openModal('bragboard');

  if (autoPrompt) {
    document.getElementById('brag-week-label').textContent = "🎉 It's Friday — here's what you accomplished this week";
  }
}

function _bragSubjectForWeek() {
  const aud = BRAG_AUDIENCE[_bragTab];
  const weekStart = document.getElementById('brag-week-start').value;
  const weekLabel = weekStart ? new Date(weekStart + 'T12:00:00').toLocaleDateString('en-US', { month:'short', day:'numeric' }) : '';
  return aud.subjectPrefix + (weekLabel ? ' — Week of ' + weekLabel : '');
}

function setBragTab(tab, btn) {
  // A genuine audience switch — resets everything, including any drafted
  // body, since the audience is what the draft was written for.
  _bragTab = tab;
  LS.set('lh_brag_last_tab', tab);
  document.querySelectorAll('#modal-bragboard .email-tab').forEach(b => b.classList.remove('active'));
  if (btn) btn.classList.add('active');
  const aud = BRAG_AUDIENCE[tab];
  document.getElementById('brag-to').value = aud.to;
  document.getElementById('brag-cc').value = '';
  _bragSubjectSnapshot = _bragSubjectForWeek();
  document.getElementById('brag-subject').value = _bragSubjectSnapshot;
  document.getElementById('brag-body').value = '';
  document.getElementById('brag-ai-status').textContent = '';
  document.getElementById('brag-copy-confirm').textContent = '';
  _bragBodySnapshot = ''; // a deliberate reset, not an edit to warn about
}

function gatherBragData() {
  const weekStartVal = document.getElementById('brag-week-start').value;
  if (!weekStartVal) return;
  const weekStart = new Date(weekStartVal + 'T00:00:00');
  const weekEnd   = new Date(weekStart); weekEnd.setDate(weekEnd.getDate() + 7);

  const inRange = (ts) => {
    if (!ts) return false;
    const d = new Date(ts);
    return d >= weekStart && d < weekEnd;
  };

  // ── Completed tasks ────────────────────────────────────────────────────────
  const completedTasks = (LS.get('lh_tasks', []) || [])
    .filter(t => t.done && inRange(t.resolvedAt))
    .map(t => ({ text: t.text, status: t.status || 'done' }));

  // ── DECA results ───────────────────────────────────────────────────────────
  const decaResults = (LS.get('lh_deca_results', []) || [])
    .filter(r => inRange(r.addedAt))
    .map(r => ({ text: `${r.student} — ${r.event}: ${r.place || 'participated'}`, notes: r.notes }));

  // ── Other organizations' results (FIXED: this was DECA-only — any other
  // organization added via Settings → Organizations never showed up in Brag
  // Board at all, since gatherBragData() only ever read lh_deca_results.
  // DECA's own section above is untouched; this is a separate, additive pull
  // from ORG_RESULTS covering every non-DECA org.) ─────────────────────────
  const orgResults = [];
  Object.keys(ORGANIZATIONS).forEach(orgId => {
    if (orgId === 'deca') return; // already covered by decaResults above
    const orgName = ORGANIZATIONS[orgId].name || orgId;
    getOrgResults(orgId)
      .filter(r => inRange(r.addedAt))
      .forEach(r => orgResults.push({ text: `${orgName}: ${r.student} — ${r.event}: ${r.place || 'participated'}`, notes: r.notes }));
  });

  // ── Trip archive ───────────────────────────────────────────────────────────
  const archivedTrips = (LS.get('lh_trip_archive', []) || [])
    .filter(a => inRange(a.archivedAt))
    .map(a => ({ text: a.trip?.name || 'Field Trip', glows: a.glows, outcome: a.outcome }));

  // ── Email bridge items (done horizon items from email source) ──────────────
  const horizon = LS.get('lh_horizon', { short:[], mid:[], long:[] });
  const emailItems = [...(horizon.short||[]), ...(horizon.mid||[]), ...(horizon.long||[])]
    .filter(h => h.source === 'email' && h.done)
    .map(h => ({ text: h.text.replace(/^📧\s*/, '') }));

  // ── Daily log ──────────────────────────────────────────────────────────────
  // Only surface genuinely brag-worthy log types — internal system events
  // (choice_picked, obs_prep_*, fieldtrip) are useful evidence but not email content
  const _bragTypes = new Set(['comms','plc_meeting','trip_archived','workday_submit']);
  const logEntries = (LS.get('lh_daily_log', []) || [])
    .filter(e => inRange(new Date(e.date || e.ts)) && _bragTypes.has(e.type))
    .map(e => ({ text: e.note || e.text || String(e) }));

  // ── E-sports match results this week ────────────────────────────────────────
  const esMatchWins = [];
  try {
    const allMatches = typeof ES_MATCHES !== 'undefined' ? ES_MATCHES : LS.get('lh_es_matches', []);
    const weekMatches = allMatches.filter(m => m.date && inRange(new Date(m.date + 'T12:00:00')));
    if (weekMatches.length) {
      const wins = weekMatches.filter(m => Number(m.us) > Number(m.them)).length;
      const losses = weekMatches.length - wins;
      weekMatches.forEach(m => {
        const won = Number(m.us) > Number(m.them);
        const game = m.game === 'ssbu' ? '🥊 Smash Bros' : m.game === 'mk8' ? '🏎️ Mario Kart' : '🎮 E-Sports';
        esMatchWins.push({ text: `${game} vs ${m.opp}: ${won ? `Win ${m.us}–${m.them}` : `Loss ${m.us}–${m.them}`}` });
      });
    }
  } catch(e) {}

  // ── Journal moments this week (proud / kindness witnessed / kindness given) ──
  // FIXED: journal reflections used to be pulled unconditionally into every
  // audience's draft (Students, Parents, Central Administration, Faculty
  // alike) with no way to exclude a specific vulnerable entry — see Say/Do
  // Ledger leader-hub finding #11. Each entry now requires an explicit
  // opt-in, remembered per entry (keyed by the journal entry's own stable
  // id + question, e.g. "j173..._q4") so a reflection isn't re-decided
  // every time the board is re-gathered within the same week. Candidates
  // are still all shown in the left panel (with a checkbox) so nothing is
  // silently hidden from review — only journalWins (the included subset)
  // ever reaches _bragData and the actual generated draft.
  const journalIncludes = LS.get('lh_brag_journal_includes', {});
  const journalCandidates = (LS.get('lh_journal_history', []) || [])
    .filter(e => inRange(e.savedAt))
    .flatMap(e => {
      const items = [];
      if ((e.q4 || '').trim()) items.push({ id: e.id + '_q4', text: (e.q4||'').trim(), kind: '🏅' });
      if ((e.q5 || '').trim()) items.push({ id: e.id + '_q5', text: (e.q5||'').trim(), kind: '👀' });
      if ((e.q6 || '').trim()) items.push({ id: e.id + '_q6', text: (e.q6||'').trim(), kind: '🤝' });
      return items;
    });
  const journalWins = journalCandidates.filter(w => journalIncludes[w.id] === true);

  // ── Manually added wins ("something we might have missed") ──────────────────
  const manualWins = (LS.get('lh_brag_manual_wins', []) || [])
    .filter(w => inRange(w.addedAt))
    .map(w => ({ text: w.text, id: w.id }));

  // ── SCR scoring sessions this week ─────────────────────────────────────────
  // Uses lh_scr_session_log written by cycleScore() — one entry per day per course
  const scrWins = [];
  try {
    const scrLog = LS.get('lh_scr_session_log', []).filter(e => inRange(e.ts));
    // Group by course for clean display
    const scrByCourse = {};
    scrLog.forEach(e => { scrByCourse[e.courseName] = (scrByCourse[e.courseName] || 0) + 1; });
    Object.entries(scrByCourse).forEach(([cname, days]) => {
      scrWins.push({ text: `SCR scores updated — ${cname} (${days} session${days !== 1 ? 's' : ''} this week)` });
    });
  } catch(e) {}

  // ── WBL hours logged this week ──────────────────────────────────────────────
  // Uses lh_wbl_hours_log written by wblLogHours() + qlLogHours()
  const wblWins = [];
  try {
    const wblLog = LS.get('lh_wbl_hours_log', []).filter(e => inRange(e.ts));
    if (wblLog.length) {
      const totalAdded = wblLog.reduce((s, e) => s + Number(e.added || 0), 0);
      const stuSet = new Set(wblLog.map(e => e.student));
      wblWins.push({ text: `WBL: +${Math.round(totalAdded * 10) / 10}h logged for ${stuSet.size} student${stuSet.size !== 1 ? 's' : ''} this week` });
    }
  } catch(e) {}

  _bragData = { completedTasks, decaResults, orgResults, trips: archivedTrips, emailItems, logEntries, scrWins, wblWins, esMatchWins, journalWins, manualWins };

  // Render left panel
  const renderList = (sectionId, listId, items, fmt) => {
    const sec = document.getElementById(sectionId);
    const lst = document.getElementById(listId);
    if (!items.length) { sec.style.display = 'none'; return; }
    sec.style.display = 'block';
    // escH() wraps the whole formatted line (emoji prefix + i.text) rather
    // than each call site's i.text individually — fmt() never emits real
    // markup of its own, so escaping the finished string is safe and
    // closes every field (task text, DECA result summary, trip name,
    // email subject, log note, SCR/WBL/E-Sports summary) this one shared
    // renderer touches, in one place.
    lst.innerHTML = items.map(i =>
      `
${escH(fmt(i))}
`
    ).join('');
  };

  renderList('brag-section-tasks', 'brag-list-tasks', completedTasks,  i => `${i.status === 'ontime' ? '✅' : i.status === 'late' ? '⏰' : '✔'} ${i.text}`);
  renderList('brag-section-deca',  'brag-list-deca',  decaResults,     i => `🏆 ${i.text}`);
  renderList('brag-section-trips', 'brag-list-trips', archivedTrips,   i => `🚌 ${i.text}`);
  renderList('brag-section-email', 'brag-list-email', emailItems,      i => `📧 ${i.text}`);
  renderList('brag-section-log',   'brag-list-log',   logEntries,      i => `📋 ${i.text}`);
  renderList('brag-section-scr',   'brag-list-scr',   scrWins,         i => `📊 ${i.text}`);
  renderList('brag-section-wbl',     'brag-list-wbl',     wblWins,         i => `⏱ ${i.text}`);
  renderList('brag-section-esports', 'brag-list-esports', esMatchWins,     i => `🎮 ${i.text}`);

  // Journal moments get an opt-in checkbox instead of the shared renderList
  // helper — every candidate for the week is shown (nothing hidden from
  // review), but only checked ones count toward totalWins/join _bragData.
  const journalSec = document.getElementById('brag-section-journal');
  const journalLst = document.getElementById('brag-list-journal');
  if (!journalCandidates.length) {
    journalSec.style.display = 'none';
  } else {
    journalSec.style.display = 'block';
    journalLst.innerHTML = journalCandidates.map(w => {
      const included = journalIncludes[w.id] === true;
      return `

        
        ${w.kind} ${escH(w.text)}
      
`;
    }).join('');
  }

  // Manual wins get delete buttons — rendered separately from the shared renderList helper
  const manualSec = document.getElementById('brag-section-manual');
  const manualLst = document.getElementById('brag-list-manual');
  if (!manualWins.length) {
    manualSec.style.display = 'none';
  } else {
    manualSec.style.display = 'block';
    manualLst.innerHTML = manualWins.map(w =>
      `

        ✍️ ${escH(w.text)}
        
      
`
    ).join('');
  }

  const totalWins = completedTasks.length + decaResults.length + archivedTrips.length + emailItems.length + logEntries.length + scrWins.length + wblWins.length + esMatchWins.length + journalWins.length + manualWins.length;
  document.getElementById('brag-empty-state').style.display = totalWins ? 'none' : 'block';

  // Refresh the subject line's week label only — this used to call
  // setBragTab(), which also wipes the "to"/"cc"/body fields, so adding a
  // manual win or changing the week silently erased an already-drafted or
  // AI-generated email with no warning. FIXED: this refresh itself was
  // still unconditional — a subject the user had hand-edited (e.g. to
  // name a specific event) got silently replaced with the generic
  // template on the next re-gather/add-win/week-change. Only overwrite
  // while the field still holds whatever this code last put there.
  const subjEl = document.getElementById('brag-subject');
  if (subjEl && subjEl.value === _bragSubjectSnapshot) {
    _bragSubjectSnapshot = _bragSubjectForWeek();
    subjEl.value = _bragSubjectSnapshot;
  }
}



// Toggles one journal reflection's include-in-Brag-Board decision. Keyed by
// the journal entry's own stable id + question (see gatherBragData()'s
// journalCandidates) so the decision survives a re-gather within the same
// week instead of resetting to the opt-in default every time.
function toggleBragJournalInclude(id, included) {
  const map = LS.get('lh_brag_journal_includes', {});
  if (included) map[id] = true; else delete map[id];
  LS.set('lh_brag_journal_includes', map);
  gatherBragData();
}

function addBragManualWin() {
  const inp = document.getElementById('brag-manual-input');
  const text = (inp?.value || '').trim();
  if (!text) return;
  const wins = LS.get('lh_brag_manual_wins', []);
  wins.unshift({ id: 'bm' + Date.now(), text, addedAt: new Date().toISOString() });
  LS.set('lh_brag_manual_wins', wins.slice(0, 200)); // plenty of headroom, avoid unbounded growth
  if (inp) inp.value = '';
  gatherBragData();
}

function deleteBragManualWin(id) {
  const all = LS.get('lh_brag_manual_wins', []);
  const removed = all.find(w => w.id === id);
  LS.set('lh_brag_manual_wins', all.filter(w => w.id !== id));
  gatherBragData();
  if (removed) {
    _undoToast(`"${(removed.text||'').slice(0,40)}" removed`, () => {
      LS.set('lh_brag_manual_wins', [...LS.get('lh_brag_manual_wins', []), removed]);
      gatherBragData();
    });
  }
}

function copyBragEmail() {
  const subject = document.getElementById('brag-subject').value;
  const body    = document.getElementById('brag-body').value;
  if (!body) return;
  const full = subject + '\n\n' + body;
  const el = document.getElementById('brag-copy-confirm');
  navigator.clipboard.writeText(full).then(() => {
    el.textContent = '✓ Copied to clipboard';
    setTimeout(() => { el.textContent = ''; }, 2500);
  }).catch(() => {
    try {
      const ta = document.createElement('textarea');
      ta.value = full;
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
      el.textContent = '✓ Copied to clipboard';
    } catch (e) {
      el.textContent = '⚠ Could not copy — select and copy the text manually';
    }
    setTimeout(() => { el.textContent = ''; }, 3500);
  });
}

// ── Friday afternoon auto-prompt ──────────────────────────────────────────────
// Fires at 2:00–2:15 PM on Fridays, or as a catch-up any time later the
// same Friday if the tab was closed/backgrounded through the window
// (matches checkCron's own missedWindow pattern) — once per week.
// Stacked below Workday reminder (z-index 99997).
function checkBragBoardPrompt() {
  const now = new Date();
  if (now.getDay() !== 5) return;                         // Fridays only
  if (!isSchoolDay(now)) return;                          // Skip on break Fridays

  const windowStart = new Date(now); windowStart.setHours(14, 0, 0, 0);
  const windowEnd   = new Date(now); windowEnd.setHours(14, 15, 0, 0);
  const withinWindow = now >= windowStart && now < windowEnd;
  const missedWindow = now >= windowEnd;
  if (!(withinWindow || missedWindow)) return;

  const today = now.toDateString();
  if (LS.get('lh_brag_prompt_date', '') === today) return;
  LS.set('lh_brag_prompt_date', today);
  showBragBoardPromptBanner();
  _fireNotification('🏆 Friday — Brag About Your Week', 'You did a lot this week. Generate a quick wins email for Ms. Green, Admin, or DECA parents.');
}
